# ----------------------------------------------------------------------------- quantum model
def _ry(th):
    c, s = torch.cos(th / 2), torch.sin(th / 2)
    return torch.stack([torch.stack([c, -s], -1), torch.stack([s, c], -1)], -2).to(torch.complex128)
def _rx(th):
    c, s = torch.cos(th / 2), torch.sin(th / 2)
    z = torch.zeros_like(c)
    re = torch.stack([torch.stack([c, z], -1), torch.stack([z, c], -1)], -2)
    im = torch.stack([torch.stack([z, -s], -1), torch.stack([-s, z], -1)], -2)
    return torch.complex(re, im)
def _rz(th):
    e = torch.exp(-0.5j * th.to(torch.complex128)); z = torch.zeros_like(e)
    return torch.stack([torch.stack([e, z], -1), torch.stack([z, e.conj()], -1)], -2)

class VQC(torch.nn.Module):
    """Data re-uploading circuit: per layer, per qubit RY(a*u) RX(b*v) then RZ RY RZ; CZ ring; <Z_k> outputs."""
    def __init__(self, n_q=6, L=4, n_out=2, seed=0, n_in=2):
        super().__init__()
        g = torch.Generator().manual_seed(seed)
        self.n_q, self.L, self.n_out, self.n_in = n_q, L, n_out, n_in
        self.enc = torch.nn.Parameter(torch.randn(L, n_q, n_in, generator=g))
        self.rot = torch.nn.Parameter(torch.rand(L, n_q, 3, generator=g) * 2 * math.pi)
        self.out_a = torch.nn.Parameter(torch.ones(n_out)); self.out_b = torch.nn.Parameter(torch.zeros(n_out))
        idx = torch.arange(2 ** n_q)
        bits = ((idx[:, None] >> (n_q - 1 - torch.arange(n_q))[None, :]) & 1)       # qubit 0 = most significant
        self.register_buffer('zsign', (1 - 2 * bits).to(torch.float64))               # (2^n, n_q)
        cz = torch.ones(2 ** n_q)
        pairs = [(q, (q + 1) % n_q) for q in range(n_q)] if n_q > 2 else ([(0, 1)] if n_q == 2 else [])
        for q1, q2 in pairs:
            cz = cz * (1 - 2 * (bits[:, q1] * bits[:, q2]).to(torch.float64))
        self.register_buffer('cz', cz.to(torch.complex128))
    def n_params(self):
        return sum(p.numel() for p in self.parameters())
    def _apply(self, psi, q, M):
        B = psi.shape[0]; n = self.n_q
        p = psi.reshape(B, 2 ** q, 2, 2 ** (n - q - 1))
        if M.dim() == 2:
            M = M.expand(B, 2, 2)
        return torch.matmul(M[:, None, :, :], p).reshape(B, 2 ** n)
    def forward(self, u, v, phi=None):
        B = u.shape[0]
        psi = torch.zeros(B, 2 ** self.n_q, dtype=torch.complex128); psi[:, 0] = 1.0
        for l in range(self.L):
            for q in range(self.n_q):
                # fused single-qubit gate: RZ(r3) RY(r2) RZ(r1) [RY(c phi)] RX(b v) RY(a u)   (rightmost acts first)
                G = _rz(self.rot[l, q, 2].expand(B)) @ _ry(self.rot[l, q, 1].expand(B)) @ _rz(self.rot[l, q, 0].expand(B))
                if self.n_in == 3:
                    G = G @ _ry(self.enc[l, q, 2] * phi)
                G = G @ _rx(self.enc[l, q, 1] * v) @ _ry(self.enc[l, q, 0] * u)
                psi = self._apply(psi, q, G)
            psi = psi * self.cz
        prob = (psi.real ** 2 + psi.imag ** 2)
        z = prob @ self.zsign[:, :self.n_out]                                            # (B, n_out)
        return z * self.out_a + self.out_b

# ----------------------------------------------------------------------------- classical models
class MLP(torch.nn.Module):
    def __init__(self, width=8, depth=2, n_out=2, seed=0, n_in=2):
        super().__init__(); torch.manual_seed(seed)
        layers, d = [], n_in
        for _ in range(depth):
            layers += [torch.nn.Linear(d, width), torch.nn.Tanh()]; d = width
        layers += [torch.nn.Linear(d, n_out)]
        self.net = torch.nn.Sequential(*layers)
    def n_params(self): return sum(p.numel() for p in self.parameters())
    def forward(self, u, v, phi=None):
        return self.net(torch.stack([u, v] + ([phi] if phi is not None else []), -1))

class FFMLP(torch.nn.Module):
    """Random Fourier features (fixed) followed by an MLP."""
    def __init__(self, m=8, width=6, depth=1, n_out=2, seed=0, scale=1.0, n_in=2):
        super().__init__(); g = torch.Generator().manual_seed(seed)
        self.register_buffer('Bf', torch.randn(n_in, m, generator=g) * scale * math.pi)
        self.mlp = MLP(width=width, depth=depth, n_out=n_out, seed=seed, n_in=2 * m)
    def n_params(self): return sum(p.numel() for p in self.parameters() if p.requires_grad)
    def forward(self, u, v, phi=None):
        z = torch.stack([u, v] + ([phi] if phi is not None else []), -1) @ self.Bf
        return self.mlp.net(torch.cat([torch.cos(z), torch.sin(z)], -1))

def matched_mlp(target, depth=2, seed=0, n_in=2):
    best = None
    for w in range(2, 400):
        n = MLP(w, depth, seed=seed, n_in=n_in).n_params()
        if best is None or abs(n - target) < abs(best[1] - target): best = (w, n)
    return MLP(best[0], depth, seed=seed, n_in=n_in)

def matched_ff(target, m=8, depth=1, seed=0, n_in=2):
    best = None
    for w in range(2, 400):
        n = FFMLP(m, w, depth, seed=seed, n_in=n_in).n_params()
        if best is None or abs(n - target) < abs(best[1] - target): best = (w, n)
    return FFMLP(m, best[0], depth, seed=seed, n_in=n_in)

# ----------------------------------------------------------------------------- fields and losses
class Fields:
    """Network outputs -> physical fields with hard initial conditions.
    param='sigma' (default): T = t*y0, sigma = t*y1  (sigma_t(x,0) = y1(x,0) can match -gamma e^{-x} f(0+)).
    param='w':               T = t*y0, w = t^2*y1, sigma = w - gamma*T  (superseded: the gamma*T cancellation
                             made sigma errors feed back into T; kept for the record)."""
    def __init__(self, model, gamma, case, param='sigma', front=False, kappa=3.0):
        self.m, self.gamma, self.case, self.param, self.front, self.kappa = model, gamma, case, param, front, kappa
        self.f0 = float(gt.pulse(np.array([0.0]), case)[0])   # f(0+)
    def sigma(self, x, t):
        u = 2 * x / XMAX - 1; v = 2 * t / TMAX - 1
        if self.front:   # smooth characteristic coordinate: which side of the wave front x = sqrt(gamma) t
            phi = torch.tanh(self.kappa * (t - x / math.sqrt(self.gamma)))
            y = self.m(u, v, phi)
        else:
            y = self.m(u, v)
        if self.param == 'taylor':
            # hard initial conditions including the initial rates:  T_t(x,0) = e^{-x} f(0+)  (heat eq. at t=0)
            # and sigma_t(x,0+) = -gamma e^{-x} f(0+)  (impulsive launch of the stress wave); the network learns the O(t^2) rest.
            ex = torch.exp(-x) * self.f0
            return t * ex + t ** 2 * y[:, 0], -self.gamma * t * ex + t ** 2 * y[:, 1]
        T = t * y[:, 0]
        if self.param == 'sigma':
            # sigma = t y1 (unscaled; the gamma-scaled variant sigma = gamma t y1 started the circuits far from the
            # solution and trained worse at equal budget, see STATUS.md 20 Oct)
            return T, t * y[:, 1]
        return T, t ** 2 * y[:, 1] - self.gamma * T
    def __call__(self, x, t):
        T, S = self.sigma(x, t); return T, S + self.gamma * T

def grad(y, x):
    return torch.autograd.grad(y, x, torch.ones_like(y), create_graph=True)[0]

def physics_loss(F, xc, tc, tb, Tb6, Sb6, weights=(1.0, 1.0, 1.0), w_scale=None, ic_weight=1.0, ic_xmin=0.3):
    xc = xc.requires_grad_(True); tc = tc.requires_grad_(True)
    T, S = F.sigma(xc, tc)
    T_t = grad(T, tc); T_tt = grad(T_t, tc); T_x = grad(T, xc); T_xx = grad(T_x, xc)
    S_t = grad(S, tc); S_tt = grad(S_t, tc); S_x = grad(S, xc); S_xx = grad(S_x, xc)
    f = torch.as_tensor(gt.pulse(tc.detach().numpy(), F.case))
    r_h = T_t - T_xx - torch.exp(-xc) * f                       # heat equation
    r_w = S_xx - S_tt / F.gamma - T_tt                           # stress wave equation, eq. (15) form
    if w_scale is not None: r_w = r_w / w_scale
    # surface x = 0
    x0 = torch.zeros_like(tb).requires_grad_(True); t0 = tb.clone()
    T0, s0 = F.sigma(x0, t0); T0_x = grad(T0, x0)
    # far boundary x = XMAX (exact Dirichlet data)
    x6 = torch.full_like(tb, XMAX); T6, s6 = F.sigma(x6, tb)
    # initial rates at t = 0 (soft):  T_t(x,0) = e^{-x} f(0+),  sigma_t(x,0+) = -gamma e^{-x} f(0+)
    # (with T = t*y0, sigma = t*y1 these are conditions on y(x, t=0); the sigma rate is what makes the stress problem well posed)
    li = torch.zeros(())
    if F.param == 'sigma' and ic_weight > 0:
        # the exact rates are discontinuous at the corner (x, t) = (0, 0) (free surface), so the rate condition is
        # imposed only for x >= ic_xmin
        xi = xc.detach(); xi = xi[xi >= ic_xmin]
        yi = F.m(2 * xi / XMAX - 1, torch.full_like(xi, -1.0), *([torch.tanh(F.kappa * (-xi / math.sqrt(F.gamma)))] if F.front else []))
        ex = torch.exp(-xi) * F.f0
        li = ic_weight * (((yi[:, 0] - ex) ** 2).mean() + (((yi[:, 1] + F.gamma * ex) / F.gamma) ** 2).mean())
    lp = (r_h ** 2).mean() + (r_w ** 2).mean()
    lb = (T0_x ** 2).mean() + (s0 ** 2).mean() + ((T6 - Tb6) ** 2).mean() + ((s6 - Sb6) ** 2).mean() + li
    return weights[0] * lp + weights[1] * lb, dict(pde=lp.item(), bc=lb.item(), ic=float(li.detach()))

def data_loss(F, xd, td, Td, Sd):
    T, S = F.sigma(xd, td)
    return ((T - Td) ** 2).mean() + ((S - Sd) ** 2).mean()

# ----------------------------------------------------------------------------- evaluation
def evaluate(F, gamma, case, n=241):
    X, Tt = gt.test_grid(n)
    Te, Se = gt.exact(X, Tt, case, gamma)
    with torch.no_grad():
        T, S = F.sigma(torch.as_tensor(X.ravel()), torch.as_tensor(Tt.ravel()))
    T = T.numpy().reshape(X.shape); S = S.numpy().reshape(X.shape)
    return dict(relL2_T=gt.rel_l2(T, Te), relL2_sigma=gt.rel_l2(S, Se),
                maxabs_T=float(np.abs(T - Te).max()), maxabs_sigma=float(np.abs(S - Se).max()))

# ----------------------------------------------------------------------------- training
def sample_colloc(rng, n, mode='mixed'):
    """Collocation sampling.  'uniform': uniform on the domain.  'mixed': half uniform, half concentrated
    near the heated corner and the early times (x ~ Exp(1), t ~ Exp(0.7), truncated), where the source and
    the wave launch live.  The physics loss is then a weighted residual integral; the weighting is declared."""
    if mode == 'uniform':
        return rng.uniform(0, XMAX, n), rng.uniform(1e-3, TMAX, n)
    n1 = n // 2; n2 = n - n1
    x1 = rng.uniform(0, XMAX, n1); t1 = rng.uniform(1e-3, TMAX, n1)
    x2 = np.minimum(rng.exponential(1.0, n2), XMAX); t2 = np.clip(rng.exponential(0.7, n2), 1e-3, TMAX)
    return np.concatenate([x1, x2]), np.concatenate([t1, t2])

def train(model, case, gamma, regime='physics', iters=3000, lr=5e-3, n_c=1024, n_b=128, n_d=256,
          seed=0, log_every=500, val_points=2000, verbose=True, sampling='mixed', weights=(1.0, 1.0, 1.0),
          w_scale=None, lbfgs_iters=0, front=False, param='sigma', ic_weight=1.0, ic_xmin=0.3):
    rng = np.random.default_rng(seed); torch.manual_seed(seed)
    F = Fields(model, gamma, case, param=param, front=front)
    # fixed far-boundary data and validation/data sets from the exact solution
    tbb = rng.uniform(1e-3, TMAX, 4096); Tb_all, Sb_all = gt.exact(np.full_like(tbb, XMAX), tbb, case, gamma)
    xv = rng.uniform(0, XMAX, val_points); tv = rng.uniform(1e-3, TMAX, val_points); Tv, Sv = gt.exact(xv, tv, case, gamma)
    xd = rng.uniform(0, XMAX, n_d); td = rng.uniform(1e-3, TMAX, n_d); Td, Sd = gt.exact(xd, td, case, gamma)
    xd, td, Td, Sd = map(torch.as_tensor, (xd, td, Td, Sd))
    opt = torch.optim.Adam(model.parameters(), lr=lr)
    sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, iters, eta_min=lr * 0.05)
    t0 = time.time(); hist = []; gnorms = []
    for it in range(1, iters + 1):
        opt.zero_grad()
        parts = {}
        if regime in ('physics', 'combined'):
            xc_np, tc_np = sample_colloc(rng, n_c, sampling); xc = torch.as_tensor(xc_np); tc = torch.as_tensor(tc_np)
            ib = rng.integers(0, 4096, n_b)
            tb = torch.as_tensor(tbb[ib]); Tb6 = torch.as_tensor(Tb_all[ib]); Sb6 = torch.as_tensor(Sb_all[ib])
            loss, parts = physics_loss(F, xc, tc, tb, Tb6, Sb6, weights, w_scale, ic_weight, ic_xmin)
        else:
            loss = torch.zeros(())
        if regime in ('data', 'combined'):
            ld = data_loss(F, xd, td, Td, Sd); parts['data'] = ld.item(); loss = loss + ld
        loss.backward()
        gn = math.sqrt(sum((p.grad ** 2).sum().item() for p in model.parameters() if p.grad is not None))
        gnorms.append(gn)
        opt.step(); sched.step()
        if it % log_every == 0 or it == iters:
            with torch.no_grad():
                T, S = F.sigma(torch.as_tensor(xv), torch.as_tensor(tv))
                val = gt.rel_l2(T.numpy(), Tv) + gt.rel_l2(S.numpy(), Sv)
            hist.append(dict(it=it, loss=loss.item(), val=val, **parts))
            if verbose: print("  it %5d loss %.3e val %.4f %s  (%.0fs)" % (it, loss.item(), val, parts, time.time() - t0), flush=True)
    if lbfgs_iters > 0 and regime in ('physics', 'combined'):
        xf_np, tf_np = sample_colloc(rng, 4 * n_c, sampling); xf = torch.as_tensor(xf_np); tf = torch.as_tensor(tf_np)
        ibf = rng.integers(0, 4096, 4 * n_b); tbf = torch.as_tensor(tbb[ibf]); Tbf = torch.as_tensor(Tb_all[ibf]); Sbf = torch.as_tensor(Sb_all[ibf])
        lb = torch.optim.LBFGS(model.parameters(), lr=1.0, max_iter=lbfgs_iters, max_eval=int(lbfgs_iters * 1.25),
                               tolerance_grad=1e-12, tolerance_change=1e-14, history_size=50, line_search_fn='strong_wolfe')
        state = dict(n=0)
        def closure():
            lb.zero_grad()
            loss, _ = physics_loss(F, xf.clone(), tf.clone(), tbf, Tbf, Sbf, weights, w_scale, ic_weight, ic_xmin)
            if regime == 'combined': loss = loss + data_loss(F, xd, td, Td, Sd)
            loss.backward(); state['n'] += 1
            return loss
        lb.step(closure)
        with torch.no_grad():
            T, S = F.sigma(torch.as_tensor(xv), torch.as_tensor(tv))
            val = gt.rel_l2(T.numpy(), Tv) + gt.rel_l2(S.numpy(), Sv)
        hist.append(dict(it=iters, lbfgs_evals=state['n'], val=val))
        if verbose: print("  lbfgs %d evals  val %.4f  (%.0fs)" % (state['n'], val, time.time() - t0), flush=True)
    res = evaluate(F, gamma, case)
    res.update(dict(train_s=time.time() - t0, iters=iters, n_params=model.n_params(), val=hist[-1]['val'],
                    hist=hist, gradnorm_first=float(np.mean(gnorms[:50])), gradnorm_last=float(np.mean(gnorms[-50:]))))
    return res

def make_model(kind, n_q=6, L=4, target=None, seed=0, n_in=2):
    if kind == 'vqc': return VQC(n_q, L, seed=seed, n_in=n_in)
    if kind == 'mlp': return matched_mlp(target, seed=seed, n_in=n_in)
    if kind == 'ff': return matched_ff(target, seed=seed, n_in=n_in)
    if kind == 'mlpref': return MLP(64, 4, seed=seed, n_in=n_in)
    raise ValueError(kind)

# ----------------------------------------------------------------------------- E2: gradient variance at init
def grad_variance(n_q, L, loss_type, case='exp', gamma=9.0, n_init=200, n_pts=256, seed=0):
    rng = np.random.default_rng(seed)
    xc = torch.as_tensor(rng.uniform(0, XMAX, n_pts)); tc = torch.as_tensor(rng.uniform(1e-3, TMAX, n_pts))
    tb = torch.as_tensor(rng.uniform(1e-3, TMAX, 64)); Tb6, Sb6 = gt.exact(np.full(64, XMAX), tb.numpy(), case, gamma)
    Tb6, Sb6 = torch.as_tensor(Tb6), torch.as_tensor(Sb6)
    Td, Sd = gt.exact(xc.numpy(), tc.numpy(), case, gamma); Td, Sd = torch.as_tensor(Td), torch.as_tensor(Sd)
    gs = []
    for i in range(n_init):
        m = VQC(n_q, L, seed=10000 * seed + i); F = Fields(m, gamma, case)
        if loss_type == 'data': loss = data_loss(F, xc, tc, Td, Sd)
        elif loss_type == 'physics': loss, _ = physics_loss(F, xc.clone(), tc.clone(), tb, Tb6, Sb6)
        else:
            lp, _ = physics_loss(F, xc.clone(), tc.clone(), tb, Tb6, Sb6); loss = lp + data_loss(F, xc, tc, Td, Sd)
        g = torch.autograd.grad(loss, m.rot)[0][0, 0, 1].item()      # first-layer RY angle on qubit 0
        gs.append(g)
    gs = np.array(gs)
    return dict(n_q=n_q, L=L, loss=loss_type, var=float(gs.var()), mean_abs=float(np.abs(gs).mean()), n_init=n_init)

===== FILE: ground_truth.py =====
"""ground_truth.py -- exact solutions of the nondimensional laser-pulse thermoelastic problem
(one-dimensional, semi-infinite solid, Beer-Lambert volumetric source, one-way coupled stress wave).

    T_t = T_xx + e^{-x} f(t),                T_x(0,t) = 0,   T(x,0) = 0,            x >= 0
    sigma_xx - (1/gamma) sigma_tt = T_tt,     sigma(0,t) = 0, sigma(x,0) = 0,
    sigma_t(x,0+) = -gamma e^{-x} f(0+)       (all fields identically zero for t < 0; this is the
                                              initial rate implied by u_t = 0 and the constitutive law)

Pulses:   exp     f = Ae exp(-beta t)
          step    f = As [H(t) - H(t - Delta)]
          smooth  f = A [exp(-beta1 t) - exp(-beta2 t)]      (f(0+) = 0: no impulsive wave launch)

The closed forms below were obtained by Laplace transform (residues + standard erfc tables).  They are
NOT the formulas printed in Al-Qahtani, Lasers in Engineering (2026), eqs. (11), (12), (19)-(26), which
contain sign/coefficient errors (they violate T(x,0) = 0 and grow like exp(gamma t)).  Verification scope (log: verification_log_closed_form.txt): (i) against a 30-digit de Hoog numerical inverse
Laplace transform the corrected forms agree to the printed 7 significant digits at 25 (x, t) points per pulse
for the exponential and the rectangular pulse (gamma = 9), and the smooth-pulse superposition agrees with the
direct inversion of its transform to 5.1e-16 at 5 points; (ii) against a method-of-lines finite-difference
solution (RK4, dx = 0.01) the maxima of |closed form - FD| over the same 25 points are T 4.20e-6 / sigma 6.18e-3
(exponential) and T 5.91e-6 / sigma 6.19e-3 (rectangular), the sigma value being the FD discretisation error at
the wave front, not a closed-form error; (iii) the self-test below regresses five points against stored 30-digit
inversion values (worst deviation 4.0e-10).  These are pointwise checks, not a uniform accuracy bound.
Only numpy and scipy are required.  Run `python ground_truth.py` for the self-test.
"""
import numpy as np
from scipy.special import erfc, erfcx, dawsn

# ----------------------------------------------------------------------------- helpers
def _xi(x, t):
    return x / (2.0 * np.sqrt(t))

def _Eerfcx(xi, u):
    """Stable e^{-xi^2} erfcx(u) for real u of either sign, using erfcx(-u) = 2 e^{u^2} - erfcx(u)."""
    u = np.asarray(u, float)
    direct = np.exp(-xi**2) * erfcx(np.where(u >= 0, u, 0.0))
    alt = 2.0 * np.exp(u**2 - xi**2) - np.exp(-xi**2) * erfcx(np.where(u < 0, -u, 0.0))
    return np.where(u >= 0, direct, alt)

# ----------------------------------------------------------------------------- temperature
def T_exp(x, t, beta, Ae=1.0):
    """f = Ae e^{-beta t}.  Conventional form:
    T = Ae/(1+beta) { e^{t-x}/2 erfc((2t-x)/(2 sqrt t)) + e^{t+x}/2 erfc((2t+x)/(2 sqrt t)) - e^{-x-beta t}
                      - e^{-beta t}/sqrt(beta) Im[ e^{i sqrt(beta) x} erfc(x/(2 sqrt t) + i sqrt(beta t)) ] }"""
    x = np.asarray(x, float); t = np.asarray(t, float)
    xi = _xi(x, t); E = np.exp(-xi**2); eta = np.sqrt(beta * t)
    val = (0.5 * _Eerfcx(xi, np.sqrt(t) - xi) + 0.5 * E * erfcx(xi + np.sqrt(t))
           - np.exp(-x - beta * t)
           - E * np.imag(erfcx(xi + 1j * eta)) / np.sqrt(beta)) / (1.0 + beta)
    return Ae * val

def _U_T(x, t):
    """Unit continuous heating f = H(t):
    U_T = -e^{-x} + e^{t-x}/2 erfc((2t-x)/(2 sqrt t)) + e^{t+x}/2 erfc((2t+x)/(2 sqrt t))
          + 2 sqrt(t/pi) e^{-x^2/(4t)} - x erfc(x/(2 sqrt t))"""
    xi = _xi(x, t); E = np.exp(-xi**2)
    return (-np.exp(-x) + 0.5 * _Eerfcx(xi, np.sqrt(t) - xi) + 0.5 * E * erfcx(xi + np.sqrt(t))
            + 2.0 * np.sqrt(t / np.pi) * E - x * erfc(xi))

def T_step(x, t, As, Delta):
    x = np.asarray(x, float); t = np.asarray(t, float)
    out = As * _U_T(x, t)
    m = t > Delta
    if np.any(m):
        out = np.where(m, out - As * _U_T(x, np.where(m, t - Delta, 1.0)), out)
    return out

def T_smooth(x, t, beta1, beta2, A):
    return T_exp(x, t, beta1, A) - T_exp(x, t, beta2, A)

# ----------------------------------------------------------------------------- stress
def _sigma_unit(x, t, beta, gamma):
    """Stress for f = e^{-beta t} (beta = 0 gives the unit-step response).  With a = sqrt(gamma),
    tau = x/a, xi = x/(2 sqrt t), eta = sqrt(beta t) and
      c1 = g/((g-1)(1+b)), c2 = g/((g-1)(g+b)), c3 = -g/((1+b)(g+b)),
      r+ = -g a/(2(a-1)(a+b)), r- = g a/(2(a+1)(a-b)), rb = g b^2/((1+b)(b^2-g)),
    sigma = c1 [ -e^{-xi^2}/sqrt(pi t) + e^{t+x}/2 erfc(xi+sqrt t) + e^{t-x}/2 erfc(sqrt t - xi) ]
          + c2 [  e^{-xi^2}/sqrt(pi t) - a/2 e^{g t + a x} erfc(xi + a sqrt t) + a/2 e^{g t - a x} erfc(xi - a sqrt t) ]
          + c3 [ -e^{-xi^2}/sqrt(pi t) - sqrt(b) e^{-b t} Im( e^{i sqrt(b) x} erfc(xi + i eta) ) ]
          + rb e^{-b t - x} + r+ e^{a t - x} + r- e^{-a t - x}
          + H(t - tau) { c1 [ 1/sqrt(pi(t-tau)) - e^{t-tau} erfc(sqrt(t-tau)) ]
                        + c2 [ -1/sqrt(pi(t-tau)) - a e^{g(t-tau)} erf(a sqrt(t-tau)) ]
                        + c3 [ 1/sqrt(pi(t-tau)) - sqrt(b) e^{-b(t-tau)} erfi(sqrt(b(t-tau))) ]
                        - rb e^{-b(t-tau)} - r+ e^{a(t-tau)} - r- e^{-a(t-tau)} }.
    The implementation regroups the exponentially growing pieces (which cancel behind the wave front)
    into bounded erfcx terms.  Singular parameter combinations to avoid: gamma = 1, a = beta, beta^2 = gamma."""
    x = np.asarray(x, float); t = np.asarray(t, float)
    a = np.sqrt(gamma); xi = _xi(x, t); E = np.exp(-xi**2); st = np.sqrt(t)
    c1 = gamma / ((gamma - 1) * (1 + beta))
    c2 = gamma / ((gamma - 1) * (gamma + beta))
    c3 = -gamma / ((1 + beta) * (gamma + beta))
    r_a = -gamma * a / (2 * (a - 1) * (a + beta))
    r_ma = gamma * a / (2 * (a + 1) * (a - beta))
    r_b = gamma * beta**2 / ((1 + beta) * (beta**2 - gamma))
    tau = x / a; H = (t > tau); tm = np.where(H, t - tau, 1.0)
    spt = np.sqrt(np.pi * t); sptm = np.sqrt(np.pi * tm)
    g1 = c1 * (-E / spt + 0.5 * E * erfcx(xi + st) + 0.5 * _Eerfcx(xi, st - xi)
               + np.where(H, 1.0 / sptm - erfcx(np.sqrt(tm)), 0.0))
    ahead = c2 * (E / spt - 0.5 * a * E * erfcx(xi + a * st) + 0.5 * a * _Eerfcx(xi, xi - a * st))
    behind = c2 * (E / spt - 0.5 * a * E * erfcx(xi + a * st) - 0.5 * a * _Eerfcx(xi, a * st - xi)
                   + a * erfcx(a * np.sqrt(tm)) - 1.0 / sptm)
    g2 = np.where(H, behind, ahead)
    eta = np.sqrt(beta * t); etam = np.sqrt(beta * tm)
    g3 = (-c3 * E / spt - c3 * np.sqrt(beta) * E * np.imag(erfcx(xi + 1j * eta))
          + r_b * np.exp(-beta * t - x)
          + np.where(H, c3 / sptm - c3 * np.sqrt(beta) * (2 / np.sqrt(np.pi)) * dawsn(etam)
                     - r_b * np.exp(-beta * tm), 0.0))
    g4 = r_a * np.exp(a * t - x) * (~H) + r_ma * (np.exp(-a * t - x) - np.where(H, np.exp(x - a * t), 0.0))
    return g1 + g2 + g3 + g4

def sigma_exp(x, t, beta, gamma, Ae=1.0):
    return Ae * _sigma_unit(x, t, beta, gamma)

def sigma_step(x, t, As, Delta, gamma):
    x = np.asarray(x, float); t = np.asarray(t, float)
    out = As * _sigma_unit(x, t, 0.0, gamma)
    m = t > Delta
    if np.any(m):
        out = np.where(m, out - As * _sigma_unit(x, np.where(m, t - Delta, 1.0), 0.0, gamma), out)
    return out

def sigma_smooth(x, t, beta1, beta2, A, gamma):
    return sigma_exp(x, t, beta1, gamma, A) - sigma_exp(x, t, beta2, gamma, A)

"""noise_eval.py -- E4: hardware-realism evaluation of a trained VQC (inference only).
(a) finite shots: each <Z_k> estimated from N_shots Bernoulli samples of the exact probability (1+<Z_k>)/2;
(b) depolarising noise: single-qubit depolarising channel with probability p applied to every qubit after every
    layer, simulated exactly with a density matrix (numpy), for n_q <= 8."""
import numpy as np, torch, sys, os
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import qpinn_local as q, ground_truth as gt

def gate_matrices(model, u, v):
    """Per-point list over layers of per-qubit fused 2x2 gates (numpy complex), from the torch model."""
    with torch.no_grad():
        B = u.shape[0]; out = []
        for l in range(model.L):
            layer = []
            for k in range(model.n_q):
                G = (q._rz(model.rot[l, k, 2].expand(B)) @ q._ry(model.rot[l, k, 1].expand(B)) @ q._rz(model.rot[l, k, 0].expand(B))
                     @ q._rx(model.enc[l, k, 1] * v) @ q._ry(model.enc[l, k, 0] * u))
                layer.append(G.numpy())
            out.append(layer)
        return out

def _apply_rho(rho, k, G, n):
    # rho: (B, 2^n, 2^n); apply G on qubit k: rho -> (G_k) rho (G_k)^dagger
    B = rho.shape[0]; d = 2 ** n
    r = rho.reshape(B, 2 ** k, 2, 2 ** (n - k - 1), d)
    r = np.einsum('bij,bajcm->baicm', G, r).reshape(B, d, d)
    r = r.reshape(B, d, 2 ** k, 2, 2 ** (n - k - 1))
    r = np.einsum('bij,bmajc->bmaic', G.conj(), r).reshape(B, d, d)
    return r

def _depolarise(rho, k, p, n):
    B = rho.shape[0]; d = 2 ** n
    X = np.array([[0, 1], [1, 0]], complex); Y = np.array([[0, -1j], [1j, 0]]); Z = np.diag([1.0, -1.0]).astype(complex)
    out = (1 - p) * rho
    for P in (X, Y, Z):
        out = out + (p / 3) * _apply_rho(rho, k, np.broadcast_to(P, (B, 2, 2)), n)
    return out

def noisy_expectations(model, x, t, p=0.0, shots=None, rng=None):
    u = 2 * x / q.XMAX - 1; v = 2 * t / q.TMAX - 1
    ut, vt = torch.as_tensor(u), torch.as_tensor(v)
    n = model.n_q; B = len(x); d = 2 ** n
    gates = gate_matrices(model, ut, vt)
    psi0 = np.zeros(d, complex); psi0[0] = 1
    rho = np.broadcast_to(np.outer(psi0, psi0.conj()), (B, d, d)).copy()
    cz = model.cz.numpy()
    for l in range(model.L):
        for k in range(n):
            rho = _apply_rho(rho, k, gates[l][k], n)
        rho = rho * cz[None, :, None] * cz.conj()[None, None, :]
        if p > 0:
            for k in range(n):
                rho = _depolarise(rho, k, p, n)
    probs = np.real(np.einsum('bii->bi', rho))
    zs = model.zsign.numpy()[:, :model.n_out]
    ez = probs @ zs
    if shots is not None:
        rng = rng or np.random.default_rng(0)
        pk = (1 + ez) / 2
        ez = 2 * rng.binomial(shots, np.clip(pk, 0, 1)) / shots - 1
    a = model.out_a.detach().numpy(); b = model.out_b.detach().numpy()
    y = ez * a + b
    # parametrisation of qpinn_local.Fields(param='sigma'):  T = t y0,  sigma = t y1
    T = t * y[:, 0]; S = t * y[:, 1]
    return T, S

def evaluate_noisy(model, gamma, case, p=0.0, shots=None, n=61, seed=0):
    X, Tt = gt.test_grid(n); x = X.ravel(); t = Tt.ravel()
    Te, Se = gt.exact(X, Tt, case, gamma)
    T, S = noisy_expectations(model, x, t, p, shots, np.random.default_rng(seed))
    return dict(p=p, shots=shots, relL2_T=gt.rel_l2(T.reshape(X.shape), Te), relL2_sigma=gt.rel_l2(S.reshape(X.shape), Se))

if __name__ == '__main__':
    m = q.VQC(4, 2)
    X, Tt = gt.test_grid(21); x = X.ravel(); t = Tt.ravel()
    T0, S0 = noisy_expectations(m, x, t)
    with torch.no_grad():
        T1, S1 = q.Fields(m, 9.0, 'exp').sigma(torch.as_tensor(x), torch.as_tensor(t))
    print("density-matrix vs statevector max diff:", np.abs(T0 - T1.numpy()).max(), np.abs(S0 - S1.numpy()).max())
    print(evaluate_noisy(m, 9.0, 'exp', p=0.01, n=21), evaluate_noisy(m, 9.0, 'exp', shots=1000, n=21))

===== FILE: functional_count.py =====
#!/usr/bin/env python3
"""functional_count.py -- coordinate-perturbation audit of the circuit parameters (which parameters can affect the outputs).
Every trainable scalar of a freshly initialised VQC (seed 0) is perturbed by an additive +0.37 in its native units (radians for rotation angles) and the two outputs are re-evaluated
on 64 fixed random inputs (u, v ~ U(-1, 1)); a coordinate is 'inert' when max |delta y| < 1e-12.  Reports the nominal count,
the inert count split into encoding scales and rotation angles, the functional (output-affecting) count, and the layer/qubit
location of every inert coordinate.  Usage: python3 functional_count.py [n_q L ...]   (default: the six E3 configurations)."""
import sys, torch, qpinn_local as q
torch.set_default_dtype(torch.float64)
cfgs = [(4, 2), (4, 4), (6, 2), (6, 4), (6, 8), (8, 2)]
if len(sys.argv) > 1: cfgs = [(int(sys.argv[i]), int(sys.argv[i + 1])) for i in range(1, len(sys.argv), 2)]
g = torch.Generator().manual_seed(12345); u = torch.rand(64, generator=g) * 2 - 1; v = torch.rand(64, generator=g) * 2 - 1
for n_q, L in cfgs:
    m = q.VQC(n_q, L, seed=0)
    with torch.no_grad():
        y0 = m(u, v).clone()
        inert, nominal = [], 0
        for name, p in m.named_parameters():
            flat = p.view(-1)
            for i in range(flat.numel()):
                nominal += 1; old = flat[i].item(); flat[i] = old + 0.37
                d = (m(u, v) - y0).abs().max().item(); flat[i] = old
                if d < 1e-12: inert.append((name, i))
    enc = [(n, i) for n, i in inert if n == 'enc']; rot = [(n, i) for n, i in inert if n == 'rot']
    def loc(name, i):
        if name == 'enc': l, r = divmod(i, n_q * 2); qb, k = divmod(r, 2); return 'L%d q%d %s' % (l, qb, 'a' if k == 0 else 'b')
        if name == 'rot': l, r = divmod(i, n_q * 3); qb, k = divmod(r, 3); return 'L%d q%d theta%d' % (l, qb, k + 1)
        return name
    print('%dq%dL: nominal %d, inert %d (encoding %d, rotation %d), functional %d' % (n_q, L, nominal, len(inert), len(enc), len(rot), nominal - len(inert)))
    print('   inert:', ', '.join(loc(n, i) for n, i in inert))

===== FILE: family_equivalence.py =====
"""family_equivalence.py -- output-family equivalence of the depth-2 circuits (referee round 2, finding 2.2).

Claim tested: with Z readout on qubits 0 and 1 after a CZ ring, the last CZ layer is diagonal and cancels in the
Z expectations, and the preceding CZ ring couples the measured qubits only to their ring neighbours.  For L = 2 the
output therefore depends on the parameters of qubits {0, 1, 2, n_q - 1} only, so the 4-, 6- and 8-qubit depth-2
circuits realise the same function family up to the relabelling {0,1,2,3} <-> {0,1,2,n_q-1}.  The test transplants
the encoding scales, rotation angles and readout affine parameters of a 4-qubit circuit onto those qubits of a
6- and an 8-qubit circuit (other qubits random) and compares the outputs on 64 random inputs.  L = 4 is included
as a control: there the deeper light cone reaches every qubit and the outputs differ.
Run from analysis/:  OMP_NUM_THREADS=2 python3 family_equivalence.py
"""
import sys, torch
sys.path.insert(0, '.')
import qpinn_local as q

def transplant(src, n_q, map_q, seed):
    dst = q.VQC(n_q, src.L, seed=seed)
    with torch.no_grad():
        for s_q, d_q in enumerate(map_q):
            dst.enc[:, d_q, :] = src.enc[:, s_q, :]
            dst.rot[:, d_q, :] = src.rot[:, s_q, :]
        dst.out_a.copy_(src.out_a); dst.out_b.copy_(src.out_b)
    return dst

if __name__ == '__main__':
    torch.manual_seed(0)
    B = 64
    u = torch.rand(B, dtype=torch.float64) * 6; v = torch.rand(B, dtype=torch.float64) * 3
    print('family equivalence test: 64 random inputs (x in [0,6], t in [0,3]); float64/complex128 statevector')
    for L in (2, 4):
        src = q.VQC(4, L, seed=3)
        with torch.no_grad():
            src.out_a.copy_(torch.tensor([1.3, -0.7])); src.out_b.copy_(torch.tensor([0.2, 0.5]))
            y4 = src(u, v)
            for n_q in (6, 8):
                dst = transplant(src, n_q, [0, 1, 2, n_q - 1], seed=11)
                d = (dst(u, v) - y4).abs().max().item()
                print('L=%d: 4-qubit parameters transplanted onto qubits (0,1,2,%d) of the %d-qubit circuit: max |output difference| = %.2e %s'
                      % (L, n_q - 1, n_q, d, '(identical function family)' if d < 1e-12 else '(different: control)'))

===== FILE: closed_form_corrected.py =====
"""Re-derived closed-form solutions (Laplace transform, residues + standard erfc tables) for
    T_t = T_xx + e^{-x} f(t),  T_x(0,t)=0,  T(x,0)=0
    sigma_xx - (1/gamma) sigma_tt = T_tt,  sigma(0,t)=0,  sigma(x,0) = 0 and sigma_t(x,0+) = -gamma e^{-x} f(0+)
    (the impulsive launch of the stress wave; equivalently w = sigma + gamma T has w = w_t = 0 at t = 0)
with f = Ae e^{-beta t} (exponential pulse) or f = As [H(t) - H(t-Delta)] (step pulse).
Transform-domain solution (a = sqrt(gamma), tau = x/a):
    T^ = fh [ e^{-x}/(s-1) - e^{-sqrt(s) x}/(sqrt(s)(s-1)) ]
    S^ = fh [ A(s)(e^{-x} - e^{-s tau}) - B(s)(e^{-sqrt(s) x} - e^{-s tau}) ],
    A = gamma s^2/((s-1)(gamma-s^2)),  B = gamma sqrt(s)/((s-1)(gamma-s)).
All exponentially growing pieces are regrouped so that only bounded erfcx/erfc terms remain."""
import numpy as np
from scipy.special import erfc, erfcx, dawsn

def _xi(x, t):
    return x / (2.0 * np.sqrt(t))

def _Eerfcx(xi, u):
    """Stable e^{-xi^2} erfcx(u) for real u of either sign (erfcx(-u) = 2 e^{u^2} - erfcx(u));
    valid whenever e^{u^2 - xi^2} is bounded, which holds for every use below."""
    u = np.asarray(u, float)
    direct = np.exp(-xi**2) * erfcx(np.where(u >= 0, u, 0.0))
    alt = 2.0 * np.exp(u**2 - xi**2) - np.exp(-xi**2) * erfcx(np.where(u < 0, -u, 0.0))
    return np.where(u >= 0, direct, alt)

def T_exp(x, t, beta, Ae=1.0):
    """Corrected eq. (11).  Differs from the printed one only in the sign of the argument of the
    second erfc: erfc((2t - x)/(2 sqrt t)) instead of erfc((x - 2t)/(2 sqrt t))."""
    x = np.asarray(x, float); t = np.asarray(t, float)
    xi = _xi(x, t); E = np.exp(-xi**2); eta = np.sqrt(beta * t)
    val = (0.5 * _Eerfcx(xi, np.sqrt(t) - xi) + 0.5 * E * erfcx(xi + np.sqrt(t))
           - np.exp(-x - beta * t)
           - E * np.imag(erfcx(xi + 1j * eta)) / np.sqrt(beta)) / (1.0 + beta)
    return Ae * val

def _U_T(x, t):
    """Unit-step (continuous heating, f = H(t)) temperature."""
    xi = _xi(x, t); E = np.exp(-xi**2)
    return (-np.exp(-x) + 0.5 * _Eerfcx(xi, np.sqrt(t) - xi) + 0.5 * E * erfcx(xi + np.sqrt(t))
            + 2.0 * np.sqrt(t / np.pi) * E - x * erfc(xi))

def T_step(x, t, As, Delta):
    x = np.asarray(x, float); t = np.asarray(t, float)
    out = As * _U_T(x, t)
    m = t > Delta
    if np.any(m):
        out = np.where(m, out - As * _U_T(x, np.where(m, t - Delta, 1.0)), out)
    return out

def _sigma_unit(x, t, beta, gamma):
    """Stress for f = e^{-beta t} (beta = 0 gives the unit-step response)."""
    x = np.asarray(x, float); t = np.asarray(t, float)
    a = np.sqrt(gamma); xi = _xi(x, t); E = np.exp(-xi**2); st = np.sqrt(t)
    c1 = gamma / ((gamma - 1) * (1 + beta))
    c2 = gamma / ((gamma - 1) * (gamma + beta))
    c3 = -gamma / ((1 + beta) * (gamma + beta))
    r_a = -gamma * a / (2 * (a - 1) * (a + beta))
    r_ma = gamma * a / (2 * (a + 1) * (a - beta))
    r_b = gamma * beta**2 / ((1 + beta) * (beta**2 - gamma))
    tau = x / a; H = (t > tau); tm = np.where(H, t - tau, 1.0)   # tm only used where H
    spt = np.sqrt(np.pi * t); sptm = np.sqrt(np.pi * tm)
    # group p = 1
    g1 = c1 * (-E / spt + 0.5 * E * erfcx(xi + st) + 0.5 * _Eerfcx(xi, st - xi)
               + np.where(H, 1.0 / sptm - erfcx(np.sqrt(tm)), 0.0))
    # group p = gamma
    ahead = c2 * (E / spt - 0.5 * a * E * erfcx(xi + a * st) + 0.5 * a * _Eerfcx(xi, xi - a * st))
    behind = c2 * (E / spt - 0.5 * a * E * erfcx(xi + a * st) - 0.5 * a * _Eerfcx(xi, a * st - xi)
                   + a * erfcx(a * np.sqrt(tm)) - 1.0 / sptm)
    g2 = np.where(H, behind, ahead)
    # group p = -beta
    eta = np.sqrt(beta * t); etam = np.sqrt(beta * tm)
    g3 = (-c3 * E / spt - c3 * np.sqrt(beta) * E * np.imag(erfcx(xi + 1j * eta))
          + r_b * np.exp(-beta * t - x)
          + np.where(H, c3 / sptm - c3 * np.sqrt(beta) * (2 / np.sqrt(np.pi)) * dawsn(etam)
                     - r_b * np.exp(-beta * tm), 0.0))
    # +-a exponentials
    g4 = r_a * np.exp(a * t - x) * (~H) + r_ma * (np.exp(-a * t - x) - np.where(H, np.exp(x - a * t), 0.0))
    return g1 + g2 + g3 + g4

def sigma_exp(x, t, beta, gamma, Ae=1.0):
    return Ae * _sigma_unit(x, t, beta, gamma)

def sigma_step(x, t, As, Delta, gamma):
    x = np.asarray(x, float); t = np.asarray(t, float)
    out = As * _sigma_unit(x, t, 0.0, gamma)
    m = t > Delta
    if np.any(m):
        out = np.where(m, out - As * _sigma_unit(x, np.where(m, t - Delta, 1.0), 0.0, gamma), out)
    return out

===== FILE: laplace_ref.py =====
"""Independent reference: numerical inverse Laplace transform (mpmath) of the exact transform-domain solution.
T^(x,s) = fhat(s) [ e^{-x}/(s-1) - e^{-sqrt(s) x}/(sqrt(s)(s-1)) ]          (T_x(0,t)=0, bounded)
sigma^ = s^2 C1 (e^{-x} - e^{-k x})/(1-k^2) + s^2 C2 (e^{-sqrt(s) x} - e^{-k x})/(s-k^2),  k = s/sqrt(gamma)
with C1 = fhat/(s-1), C2 = -fhat/(sqrt(s)(s-1));  sigma(0,t)=0 (free surface), zero initial data."""
import mpmath as mp

def That(x, s, fhat):
    return fhat(s) * (mp.e**(-x) / (s - 1) - mp.e**(-mp.sqrt(s) * x) / (mp.sqrt(s) * (s - 1)))

def Shat(x, s, fhat, gamma):
    k = s / mp.sqrt(gamma)
    C1 = fhat(s) / (s - 1)
    C2 = -fhat(s) / (mp.sqrt(s) * (s - 1))
    return (s**2 * C1 * (mp.e**(-x) - mp.e**(-k * x)) / (1 - k**2)
            + s**2 * C2 * (mp.e**(-mp.sqrt(s) * x) - mp.e**(-k * x)) / (s - k**2))

def invert(Fs, t, method='talbot', dps=30):
    mp.mp.dps = dps
    return mp.invertlaplace(Fs, t, method=method)

def T_exp_ref(x, t, beta, Ae=1.0, **kw):
    fhat = lambda s: Ae / (s + beta)
    return float(invert(lambda s: That(mp.mpf(x), s, fhat), mp.mpf(t), **kw))

def sigma_exp_ref(x, t, beta, gamma, Ae=1.0, **kw):
    fhat = lambda s: Ae / (s + beta)
    return float(invert(lambda s: Shat(mp.mpf(x), s, fhat, mp.mpf(gamma)), mp.mpf(t), **kw))

def _unit_step(x, t, gamma, which, **kw):
    fhat = lambda s: 1 / s
    if which == 'T':
        return float(invert(lambda s: That(mp.mpf(x), s, fhat), mp.mpf(t), **kw))
    return float(invert(lambda s: Shat(mp.mpf(x), s, fhat, mp.mpf(gamma)), mp.mpf(t), **kw))

def T_step_ref(x, t, As, Delta, **kw):
    v = As * _unit_step(x, t, None, 'T', **kw)
    if t > Delta:
        v -= As * _unit_step(x, t - Delta, None, 'T', **kw)
    return v

def sigma_step_ref(x, t, As, Delta, gamma, **kw):
    v = As * _unit_step(x, t, gamma, 'S', **kw)
    if t > Delta:
        v -= As * _unit_step(x, t - Delta, gamma, 'S', **kw)
    return v

===== FILE: fd_ref.py =====
"""Independent reference 2: method-of-lines finite differences (RK4) for the coupled system,
using w = sigma + gamma*T so that T_tt never appears:  T_t = T_xx + e^{-x} f,  w_tt = gamma (w - gamma T)_xx.
BCs: T_x(0,t)=0, w(0,t)=gamma*T(0,t) (free surface sigma=0), far field zero.
ICs: T=w=w_t=0.  (w = sigma + gamma*T is the scaled elastic strain; u_t = 0 at t=0 gives w_t(x,0+) = 0, which is NOT a zero stress rate:
 i.e. sigma_t(x,0+) = -gamma e^{-x} f(0+), the same jump the 0- Laplace transform of T_tt produces.)"""
import numpy as np

def solve(f, f0, gamma, L=24.0, N=1200, t_end=3.0, t_out=None, cfl=0.2):
    x = np.linspace(0.0, L, N + 1); dx = x[1] - x[0]
    dt = cfl * dx**2
    nsteps = int(np.ceil(t_end / dt)); dt = t_end / nsteps
    ex = np.exp(-x)
    T = np.zeros(N + 1); w = np.zeros(N + 1); p = np.zeros(N + 1)

    def rhs(t, T, w, p):
        Txx = np.zeros_like(T)
        Txx[1:-1] = (T[2:] - 2 * T[1:-1] + T[:-2]) / dx**2
        Txx[0] = 2 * (T[1] - T[0]) / dx**2
        dT = Txx + ex * f(t); dT[-1] = 0.0
        w = w.copy(); w[0] = gamma * T[0]; w[-1] = 0.0
        wxx = np.zeros_like(w)
        wxx[1:-1] = (w[2:] - 2 * w[1:-1] + w[:-2]) / dx**2
        dp = gamma * wxx - gamma**2 * Txx; dp[0] = 0.0; dp[-1] = 0.0
        dw = p.copy(); dw[0] = gamma * dT[0]; dw[-1] = 0.0
        return dT, dw, dp

    t = 0.0; out = {}
    t_out = sorted(t_out or [])
    k_out = {int(round(tt / dt)): tt for tt in t_out}
    for n in range(nsteps):
        k1 = rhs(t, T, w, p)
        k2 = rhs(t + dt/2, T + dt/2*k1[0], w + dt/2*k1[1], p + dt/2*k1[2])
        k3 = rhs(t + dt/2, T + dt/2*k2[0], w + dt/2*k2[1], p + dt/2*k2[2])
        k4 = rhs(t + dt, T + dt*k3[0], w + dt*k3[1], p + dt*k3[2])
        T = T + dt/6*(k1[0] + 2*k2[0] + 2*k3[0] + k4[0])
        w = w + dt/6*(k1[1] + 2*k2[1] + 2*k3[1] + k4[1])
        p = p + dt/6*(k1[2] + 2*k2[2] + 2*k3[2] + k4[2])
        w[0] = gamma * T[0]; w[-1] = 0.0; T[-1] = 0.0
        t += dt
        if (n + 1) in k_out:
            out[k_out[n + 1]] = (T.copy(), (w - gamma * T).copy())
    return x, out
